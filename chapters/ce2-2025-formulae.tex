\begin{enumerate}[label=\thesubsection.\arabic*, ref=\thesubsection.\theenumi,itemsep=0pt]

\item Energy Consumption
\begin{itemize}
\item Power-Time-Energy Relation: The total energy consumption $E$ is given by the product of power $P$, operating time $t$ per day, and the number of days $N$:
\begin{align}
E = P \times t \times N \label{eq:energy_consumption}
\end{align}
where $P$ is in kilowatts (kW), $t$ is in hours per day, and $N$ is the number of days, yielding energy in kilowatt-hours (kWh).
\end{itemize}

\item Alligation Formula
\begin{itemize}
\item Mixture Pricing and Proportion: When two varieties of materials or commodities with unit costs $C_P$ and $C_Q$ (where $C_P < CP < C_Q$, with $CP$ being the cost price of the mixture) are mixed in weights $W_P$ and $W_Q$, the proportion is given by:
\begin{align}
\frac{W_P}{W_Q} = \frac{C_Q - CP}{CP - C_P} \label{eq:alligation_formula}
\end{align}
\end{itemize}

\item Heron's Formula and Triangle Area
\begin{itemize}
\item Semi-Perimeter: For a triangle with side lengths $a$, $b$, and $c$, the semi-perimeter $s$ is defined as:
\begin{align}
s = \frac{a + b + c}{2} \label{eq:semi_perimeter}
\end{align}
\item Area via Heron's Formula: The area $\Delta$ of the triangle is given by:
\begin{align}
\Delta = \sqrt{s\brak{s - a}\brak{s - b}\brak{s - c}} \label{eq:herons_formula}
\end{align}
\item Area via Base and Altitude: Alternatively, given a base $b$ and corresponding altitude $h$, the area is:
\begin{align}
\Delta = \frac{1}{2} \times b \times h \label{eq:triangle_area_base_alt}
\end{align}
\end{itemize}

\item Integration by Parts
\begin{itemize}
\item Indefinite Integral Formula: For differentiable functions $u\brak{x}$ and $v\brak{x}$, the integration by parts formula is:
\begin{align}
\int u \, dv = u v - \int v \, du \label{eq:integration_by_parts}
\end{align}
Expressed explicitly in terms of $x$:
\begin{align}
\int u\brak{x} v'\brak{x} \, dx = u\brak{x}v\brak{x} - \int v\brak{x}u'\brak{x} \, dx
\end{align}
\end{itemize}

\item Vector Calculus: Curl and Divergence of Velocity Vector
\begin{itemize}
\item Two-Dimensional Velocity Field: For a 2D velocity vector $\vec{V} = u\hat{i} + v\hat{j}$, where $u$ and $v$ are scalar components along $x$ and $y$:
\begin{enumerate}
    \item $z$-Component of Curl:
    \begin{align}
    \brak{\nabla \times \vec{V}}_z = \frac{\partial v}{\partial x} - \frac{\partial u}{\partial y} \label{eq:curl_z_component}
    \end{align}
    \item Divergence of Velocity Vector:
    \begin{align}
    \nabla \cdot \vec{V} = \frac{\partial u}{\partial x} + \frac{\partial v}{\partial y} \label{eq:divergence_velocity}
    \end{align}
\end{enumerate}
\end{itemize}

\item Laplace Transform
\begin{itemize}
\item Unilateral Definition: For causal signals starting at $t = 0$:
\begin{align}
G\brak{s} = \mathcal{L}\cbrak{g\brak{t}} = \int_{0}^{\infty} g\brak{t} e^{-st} \, dt \label{eq:laplace_def}
\end{align}
\item Derivative Properties:
\begin{align}
y'\brak{t} &\system{L} sY\brak{s} - y\brak{0} \label{eq:laplace_first_deriv} \\
y''\brak{t} &\system{L} s^2 Y\brak{s} - s y\brak{0} - y'\brak{0} \label{eq:laplace_second_deriv}
\end{align}
\item Standard Transform Pairs:
\begin{align}
u\brak{t} &\system{L} \frac{1}{s}, \quad \text{Re}\brak{s} > 0 \\
e^{at} u\brak{t} &\system{L} \frac{1}{s - a}, \quad \text{Re}\brak{s} > a \label{eq:laplace_exp_inv}
\end{align}
\end{itemize}

\item Higher-Order Derivative Test and Local Extrema
\begin{itemize}
\item Critical Points: Points where the first derivative vanishes: $f'\brak{x_0} = 0$.
\item Second Derivative Test: Let $x_0$ be a critical point of $f\brak{x}$:
\begin{enumerate}
    \item If $f''\brak{x_0} < 0$, then $f\brak{x}$ has a local maximum at $x_0$.
    \item If $f''\brak{x_0} > 0$, then $f\brak{x}$ has a local minimum at $x_0$.
    \item If $f''\brak{x_0} = 0$, the test is inconclusive, and higher-order derivatives must be evaluated.
\end{enumerate}
\end{itemize}

\item Standard Deflection Formulae and Propped Cantilever Analysis
\begin{itemize}
\item Deflection due to UDL: The maximum deflection $\delta_{UDL}$ at the free end of a cantilever of length $L$ and flexural rigidity $EI$ subjected to a uniformly distributed load of intensity $w$ over the entire span is:
\begin{align}
\delta_{UDL} = -\frac{wL^4}{8EI} \label{eq:cantilever_deflection_udl}
\end{align}
\item Deflection due to End Moment: The maximum deflection $\delta_M$ at the free end of a cantilever of length $L$ and flexural rigidity $EI$ subjected to an end moment $M$ is:
\begin{align}
\delta_M = -\frac{ML^2}{2EI} \label{eq:cantilever_deflection_moment}
\end{align}
\item Deflection due to End Load: The deflection $\delta_R$ at the free end due to a point load (or redundant reaction $R_B$) at the tip is:
\begin{align}
\delta_R = \frac{R_B L^3}{3EI} \label{eq:cantilever_deflection_load}
\end{align}
\item Propped Cantilever Compatibility: For a propped cantilever with a support settlement $\Delta$ at the prop (taking upward as positive):
\begin{align}
\delta_M + \delta_R = -\Delta \implies R_B = \frac{3M}{2L} - \frac{3EI\Delta}{L^3} \label{eq:propped_cantilever_compat}
\end{align}
\end{itemize}

\item Eigenvalues and Eigenvectors of a Matrix
\begin{itemize}
\item Characteristic Equation: For a square matrix $\vec{A}$, the eigenvalues $\lambda$ are the roots of the characteristic polynomial:
\begin{align}
P\brak{\lambda} = \mydet{\lambda \vec{I} - \vec{A}} = 0 \label{eq:char_equation}
\end{align}
\item Trace and Eigenvalue Sum: The sum of the eigenvalues equals the trace of the matrix:
\begin{align}
\sum_{i=1}^n \lambda_i = \operatorname{tr}\brak{\vec{A}} = \sum_{i=1}^n a_{ii} \label{eq:eigenvalue_trace}
\end{align}
\end{itemize}

\item Chemistry and Environmental Engineering Formulae (TOC, TKN, COD)
\begin{itemize}
\item Stoichiometric Oxidation Reaction: For an organic compound $C_aH_bO_cN_d$, the complete stoichiometric oxygen demand reaction is:
\begin{align}
C_aH_bO_cN_d + \left(\frac{4a + b - 2c - 3d}{4}\right) O_2 \longrightarrow aCO_2 + \frac{b - 3d}{2}H_2O + dNH_3 \label{eq:stoich_oxidation}
\end{align}
\item Definitions:
\begin{enumerate}
	\item {\bfseries TOC (Total Organic Carbon)}: Measures the total carbon content in the compound ($12a$ per mole).
	\item {\bfseries TKN (Total Kjeldahl Nitrogen)}: Measures organic and ammonia nitrogen content ($14d$ per mole).
	\item {\bfseries COD (Chemical Oxygen Demand)}: Measures the oxygen equivalent required for chemical oxidation of the organic matter based on the moles of $O_2$ consumed ($32 \times \text{moles of } O_2$ per mole of compound).
\end{enumerate}
\end{itemize}

\item Hydrostatic Force and Moment about Hinge
\begin{itemize}
\item Hydrostatic Force: The total hydrostatic force $F$ on a submerged plane surface of area $A$ with centroidal depth $\bar{y}$ below the free surface is:
\begin{align}
F = \rho g A \bar{y} \label{eq:hydrostatic_force}
\end{align}
\item Moment about Hinge: For a gate or plate pivoted about a hinge, the resultant moment $M_P$ due to linear hydrostatic pressure distribution $p\brak{s} = \rho g \bar{y}\brak{s}$ integrated over the surface length $L$ is:
\begin{align}
M_P = \int_{0}^{L} s \, dF = \rho g b \int_{0}^{L} s \brak{h - \frac{y\brak{s}}{s}s} \, ds \label{eq:moment_about_hinge}
\end{align}
\end{itemize}

\item Highway Engineering Widening Formulae
\begin{itemize}
\item Total Extra Widening ($W_e$): On horizontal curves, extra widening is the sum of mechanical widening ($W_m$) and psychological widening ($W_p$):
\begin{align}
W_e = W_m + W_p \label{eq:total_widening}
\end{align}
where:
\begin{align}
W_m &= \frac{n l^2}{2R} \quad \text{(Mechanical Widening)} \label{eq:mechanical_widening} \\
W_p &= \frac{V}{9.5 \sqrt{R}} \quad \text{(Psychological Widening)} \label{eq:psychological_widening}
\end{align}
with $n$ as the number of lanes, $l$ as the wheelbase of the longest vehicle, $R$ as the curve radius, and $V$ as design speed in km/h.
\end{itemize}

\item System of Linear Equations and Rouch\'{e}--Capelli Theorem
\begin{itemize}
\item Matrix Form: $\vec{A}\vec{x} = \vec{b}$.
\item Solvability via Ranks:
\begin{enumerate}
    \item No Solution (Inconsistent): $\text{rank}\brak{\vec{A}} < \text{rank}\brak{\sbrak{\vec{A} \mid \vec{b}}}$
    \item Unique Solution: $\text{rank}\brak{\vec{A}} = \text{rank}\brak{\sbrak{\vec{A} \mid \vec{b}}} = n$ (number of variables)
    \item Infinitely Many Solutions: $\text{rank}\brak{\vec{A}} = \text{rank}\brak{\sbrak{\vec{A} \mid \vec{b}}} < n$
\end{enumerate}
\end{itemize}

\item Probability and Random Variables
\begin{itemize}
\item Mutually Exclusive Events: $\Pr\brak{A \cap B} = 0$ and $\Pr\brak{A \cup B} = \Pr\brak{A} + \Pr\brak{B}$.
\item Conditional Probability: $\Pr\brak{A \mid B} = \frac{\Pr\brak{A \cap B}}{\Pr\brak{B}}$ (with $\Pr\brak{B} > 0$). If $B \subset A$, $\Pr\brak{A \mid B} = 1$.
\item Mean and Variance of Discrete Random Variables:
\begin{align}
\mu &= E[X] = \sum x_i p_i \label{eq:discrete_mean} \\
\sigma^2 &= E[X^2] - \left(E[X]\right)^2 = \sum x_i^2 p_i - \mu^2 \label{eq:discrete_variance}
\end{align}
\end{itemize}

\end{enumerate}
